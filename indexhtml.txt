  <script>
    const container = document.getElementById('canvas-container');
    const scene = new THREE.Scene();
    scene.fog = new THREE.FogExp2(0x050811, 0.018);

    const camera = new THREE.PerspectiveCamera(50, window.innerWidth / window.innerHeight, 0.1, 1000);
    const renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true });
    renderer.setSize(window.innerWidth, window.innerHeight);
    renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
    container.appendChild(renderer.domElement);

    function createParticleTexture() {
      const canvas = document.createElement('canvas');
      canvas.width = 64;
      canvas.height = 64;
      const ctx = canvas.getContext('2d');
      const grad = ctx.createRadialGradient(32, 32, 0, 32, 32, 32);
      grad.addColorStop(0, 'rgba(255, 255, 255, 1)');
      grad.addColorStop(0.3, 'rgba(255, 245, 200, 0.85)');
      grad.addColorStop(0.7, 'rgba(255, 200, 90, 0.25)');
      grad.addColorStop(1, 'rgba(0, 0, 0, 0)');
      ctx.fillStyle = grad;
      ctx.fillRect(0, 0, 64, 64);
      return new THREE.CanvasTexture(canvas);
    }

    const particleTex = createParticleTexture();
    const treeGroup = new THREE.Group();
    scene.add(treeGroup);

    // 圣诞树粒子主体
    const treeCount = 3800;
    const treeGeo = new THREE.BufferGeometry();
    const treePos = new Float32Array(treeCount * 3);
    const treeCol = new Float32Array(treeCount * 3);
    const treeSeeds = [];
    const palette = [0x1f8a46, 0x2ecc71, 0xff4757, 0xffd700, 0xffffff];

    for (let i = 0; i < treeCount; i++) {
      const h = Math.random() * 13.5;
      const ratio = h / 13.5;
      const wave = 1 + 0.16 * Math.sin(ratio * Math.PI * 9);
      const radius = Math.pow(Math.random(), 0.58) * (1 - ratio) * 5.4 * wave;
      const angle = Math.random() * Math.PI * 2;

      treePos[i * 3] = Math.cos(angle) * radius;
      treePos[i * 3 + 1] = h;
      treePos[i * 3 + 2] = Math.sin(angle) * radius;

      const color = new THREE.Color(palette[Math.floor(Math.random() * palette.length)]);
      treeCol[i * 3] = color.r;
      treeCol[i * 3 + 1] = color.g;
      treeCol[i * 3 + 2] = color.b;
      treeSeeds.push({ baseY: h, phase: Math.random() * Math.PI * 2 });
    }

    treeGeo.setAttribute('position', new THREE.BufferAttribute(treePos, 3));
    treeGeo.setAttribute('color', new THREE.BufferAttribute(treeCol, 3));
    const treePoints = new THREE.Points(
      treeGeo,
      new THREE.PointsMaterial({
        size: 0.25,
        map: particleTex,
        vertexColors: true,
        transparent: true,
        depthWrite: false,
        blending: THREE.AdditiveBlending
      })
    );
    treeGroup.add(treePoints);

    // 螺旋灯带粒子
    const spiralCount = 700;
    const spiralGeo = new THREE.BufferGeometry();
    const spiralPos = new Float32Array(spiralCount * 3);
    const spiralCol = new Float32Array(spiralCount * 3);

    for (let i = 0; i < spiralCount; i++) {
      const t = i / spiralCount;
      const y = t * 13.2;
      const r = (1 - t) * 5.6 + 0.25;
      const a = t * Math.PI * 8;
      spiralPos[i * 3] = Math.cos(a) * r;
      spiralPos[i * 3 + 1] = y;
      spiralPos[i * 3 + 2] = Math.sin(a) * r;

      const c = new THREE.Color(i % 2 === 0 ? 0xffd700 : 0xff6b81);
      spiralCol[i * 3] = c.r;
      spiralCol[i * 3 + 1] = c.g;
      spiralCol[i * 3 + 2] = c.b;
    }

    spiralGeo.setAttribute('position', new THREE.BufferAttribute(spiralPos, 3));
    spiralGeo.setAttribute('color', new THREE.BufferAttribute(spiralCol, 3));
    treeGroup.add(
      new THREE.Points(
        spiralGeo,
        new THREE.PointsMaterial({
          size: 0.32,
          map: particleTex,
          vertexColors: true,
          transparent: true,
          depthWrite: false,
          blending: THREE.AdditiveBlending
        })
      )
    );

    // 树顶星光
    const topStar = new THREE.Sprite(
      new THREE.SpriteMaterial({
        map: particleTex,
        color: 0xffe066,
        transparent: true,
        blending: THREE.AdditiveBlending
      })
    );
    topStar.position.set(0, 14.1, 0);
    topStar.scale.set(3.4, 3.4, 1);
    treeGroup.add(topStar);

    // 雪花粒子
    const snowCount = 900;
    const snowGeo = new THREE.BufferGeometry();
    const snowPos = new Float32Array(snowCount * 3);
    const snowSpeeds = [];

    for (let i = 0; i < snowCount; i++) {
      snowPos[i * 3] = (Math.random() - 0.5) * 44;
      snowPos[i * 3 + 1] = Math.random() * 26 - 2;
      snowPos[i * 3 + 2] = (Math.random() - 0.5) * 44;
      snowSpeeds.push(0.02 + Math.random() * 0.035);
    }

    snowGeo.setAttribute('position', new THREE.BufferAttribute(snowPos, 3));
    const snowPoints = new THREE.Points(
      snowGeo,
      new THREE.PointsMaterial({
        size: 0.22,
        map: particleTex,
        color: 0xffffff,
        transparent: true,
        opacity: 0.8,
        depthWrite: false,
        blending: THREE.AdditiveBlending
      })
    );
    scene.add(snowPoints);

    // 烟花效果
    const fireworks = [];
    const raycaster = new THREE.Raycaster();
    const pointer = new THREE.Vector2();
    const hitPlane = new THREE.Plane(new THREE.Vector3(0, 0, 1), 0);

    function createFirework(point) {
      const count = 75;
      const geo = new THREE.BufferGeometry();
      const pos = new Float32Array(count * 3);
      const col = new Float32Array(count * 3);
      const vels = [];
      const baseColor = new THREE.Color(palette[Math.floor(Math.random() * palette.length)]);

      for (let i = 0; i < count; i++) {
        pos[i * 3] = point.x;
        pos[i * 3 + 1] = point.y;
        pos[i * 3 + 2] = point.z;

        const theta = Math.random() * Math.PI * 2;
        const phi = Math.acos(2 * Math.random() - 1);
        const speed = 0.07 + Math.random() * 0.14;
        vels.push(
          new THREE.Vector3(
            Math.sin(phi) * Math.cos(theta) * speed,
            Math.cos(phi) * speed,
            Math.sin(phi) * Math.sin(theta) * speed
          )
        );

        const c = baseColor.clone().offsetHSL((Math.random() - 0.5) * 0.1, 0, 0.1);
        col[i * 3] = c.r;
        col[i * 3 + 1] = c.g;
        col[i * 3 + 2] = c.b;
      }

      geo.setAttribute('position', new THREE.BufferAttribute(pos, 3));
      geo.setAttribute('color', new THREE.BufferAttribute(col, 3));
      const mat = new THREE.PointsMaterial({
        size: 0.32,
        map: particleTex,
        vertexColors: true,
        transparent: true,
        opacity: 1,
        depthWrite: false,
        blending: THREE.AdditiveBlending
      });

      const points = new THREE.Points(geo, mat);
      scene.add(points);
      fireworks.push({ points, vels, life: 1 });
    }

    // 交互控制
    let isDragging = false;
    let moveDist = 0;
    let lastX = 0;
    let lastY = 0;
    let angleY = 0;
    let camY = 6.5;
    const camRadius = 24;

    renderer.domElement.addEventListener('pointerdown', (e) => {
      isDragging = true;
      moveDist = 0;
      lastX = e.clientX;
      lastY = e.clientY;
    });

    window.addEventListener('pointermove', (e) => {
      if (!isDragging) return;
      const dx = e.clientX - lastX;
      const dy = e.clientY - lastY;
      moveDist += Math.abs(dx) + Math.abs(dy);
      lastX = e.clientX;
      lastY = e.clientY;
      angleY -= dx * 0.008;
      camY = Math.max(1, Math.min(16, camY + dy * 0.035));
    });

    window.addEventListener('pointerup', (e) => {
      if (isDragging && moveDist < 8) {
        pointer.x = (e.clientX / window.innerWidth) * 2 - 1;
        pointer.y = -(e.clientY / window.innerHeight) * 2 + 1;
        raycaster.setFromCamera(pointer, camera);
        const target = new THREE.Vector3();
        hitPlane.normal.copy(camera.position).normalize();
        raycaster.ray.intersectPlane(hitPlane, target);
        if (target) createFirework(target);
      }
      isDragging = false;
    });

    window.addEventListener('resize', () => {
      camera.aspect = window.innerWidth / window.innerHeight;
      camera.updateProjectionMatrix();
      renderer.setSize(window.innerWidth, window.innerHeight);
    });

    const clock = new THREE.Clock();
    function animate() {
      requestAnimationFrame(animate);
      const t = clock.getElapsedTime();

      treeGroup.rotation.y += 0.004;
      const starScale = 3.2 + Math.sin(t * 3.2) * 0.55;
      topStar.scale.set(starScale, starScale, 1);

      const tArr = treeGeo.attributes.position.array;
      for (let i = 0; i < treeCount; i += 3) {
        const s = treeSeeds[i];
        tArr[i + 1] = s.baseY + Math.sin(t * 2 + s.phase) * 0.035;
      }
      treeGeo.attributes.position.needsUpdate = true;

      const sArr = snowGeo.attributes.position.array;
      for (let i = 0; i < snowCount; i++) {
        sArr[i * 3 + 1] -= snowSpeeds[i];
        sArr[i * 3] += Math.sin(t + i) * 0.008;
        if (sArr[i * 3 + 1] < -2) {
          sArr[i * 3 + 1] = 24;
        }
      }
      snowGeo.attributes.position.needsUpdate = true;

      for (let i = fireworks.length - 1; i >= 0; i--) {
        const fw = fireworks[i];
        fw.life -= 0.018;
        if (fw.life <= 0) {
          scene.remove(fw.points);
          fw.points.geometry.dispose();
          fw.points.material.dispose();
          fireworks.splice(i, 1);
          continue;
        }
        const pos = fw.points.geometry.attributes.position.array;
        for (let j = 0; j < fw.vels.length; j++) {
          fw.vels[j].y -= 0.0025;
          fw.vels[j].multiplyScalar(0.98);
          pos[j * 3] += fw.vels[j].x;
          pos[j * 3 + 1] += fw.vels[j].y;
          pos[j * 3 + 2] += fw.vels[j].z;
        }
        fw.points.geometry.attributes.position.needsUpdate = true;
        fw.points.material.opacity = fw.life;
      }

      camera.position.x = Math.sin(angleY) * camRadius;
      camera.position.z = Math.cos(angleY) * camRadius;
      camera.position.y += (camY - camera.position.y) * 0.08;
      camera.lookAt(0, 6.2, 0);

      renderer.render(scene, camera);
    }

    animate();
  </script>

  <script>
    (function () {
      let ctx = null;
      let playing = false;
      let timer = null;
      let idx = 0;

      const freqs = {
        D4: 293.66,
        E4: 329.63,
        F4: 349.23,
        G4: 392.00,
        A4: 440.00,
        B4: 493.88,
        C5: 523.25
      };

      const song = [
        ['D4', 1], ['G4', 1], ['G4', 0.5], ['A4', 0.5], ['G4', 0.5], ['F4', 0.5],
        ['E4', 1], ['E4', 1], ['E4', 1],
        ['A4', 1], ['A4', 0.5], ['B4', 0.5], ['A4', 0.5], ['G4', 0.5],
        ['F4', 1], ['D4', 1], ['D4', 1],
        ['B4', 1], ['B4', 0.5], ['C5', 0.5], ['B4', 0.5], ['A4', 0.5],
        ['G4', 1], ['E4', 1], ['D4', 0.5], ['D4', 0.5],
        ['E4', 1], ['A4', 1], ['F4', 1],
        ['G4', 2], [null, 1]
      ];

      function playNote(freq, dur) {
        if (!ctx || !freq) return;
        const now = ctx.currentTime;
        const osc = ctx.createOscillator();
        const overtone = ctx.createOscillator();
        const gain = ctx.createGain();
        const overGain = ctx.createGain();

        osc.type = 'sine';
        osc.frequency.setValueAtTime(freq, now);

        overtone.type = 'triangle';
        overtone.frequency.setValueAtTime(freq * 2, now);
        overGain.gain.setValueAtTime(0.045, now);
        overGain.gain.exponentialRampToValueAtTime(0.0001, now + dur * 0.9);

        gain.gain.setValueAtTime(0.001, now);
        gain.gain.linearRampToValueAtTime(0.18, now + 0.02);
        gain.gain.exponentialRampToValueAtTime(0.0001, now + dur * 1.3);

        osc.connect(gain);
        overtone.connect(overGain);
        overGain.connect(gain);
        gain.connect(ctx.destination);

        osc.start(now);
        overtone.start(now);
        osc.stop(now + dur * 1.35);
        overtone.stop(now + dur * 1.35);
      }

      function next() {
        if (!playing) return;
        const [note, beats] = song[idx];
        const beatMs = 420;
        playNote(freqs[note], (beats * beatMs) / 1000);
        idx = (idx + 1) % song.length;
        timer = setTimeout(next, beats * beatMs);
      }

      const btn = document.getElementById('musicBtn');

      function updateBtnState() {
        if (!btn) return;
        if (playing) {
          btn.classList.add('playing');
          btn.textContent = '🎶 音乐播放中';
        } else {
          btn.classList.remove('playing');
          btn.textContent = '🎵 已静音';
        }
      }

      function startMusic() {
        const AudioCtx = window.AudioContext || window.webkitAudioContext;
        if (!ctx && AudioCtx) {
          ctx = new AudioCtx();
        }
        if (ctx && ctx.state === 'suspended') {
          ctx.resume();
        }
        if (playing) return;
        playing = true;
        updateBtnState();
        next();
      }

      function stopMusic() {
        playing = false;
        clearTimeout(timer);
        updateBtnState();
      }

      if (btn) {
        btn.addEventListener('click', (e) => {
          e.stopPropagation();
          if (playing) {
            stopMusic();
          } else {
            startMusic();
          }
        });
      }

      window.addEventListener('pointerdown', startMusic, { once: true });
    })();
  </script>
